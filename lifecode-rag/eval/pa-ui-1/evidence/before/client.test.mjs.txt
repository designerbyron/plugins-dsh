/**
 * Contrato estático del módulo de cliente: sin imports del harness, con los slots reales y con la
 * misma ruta de API que el servidor. No sustituye a una prueba en el host (M6), pero impide que la
 * pieza se rompa en silencio.
 */

import test from 'node:test';
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';

import { API_PREFIX } from '../lib/server.mjs';

const client = readFileSync(fileURLToPath(new URL('../client.js', import.meta.url)), 'utf8');

test('el cliente no importa paquetes del harness', () => {
  assert.equal(/from\s+['"]@deepseek-ai\//.test(client), false);
  assert.equal(/\brequire\(['"]react['"]\)/.test(client), true, 'React llega por el cargador del host');
  assert.equal(client.includes('window.__ModuleLoader__.load'), true);
});

test('registra la pestaña con los ids reales de slot', () => {
  assert.equal(client.includes("const PANEL_ID = 'lifecode-rag'"), true);
  assert.equal(client.includes("const SIDEBAR_SLOT = 'sidebar.panellist'"), true);
  assert.equal(client.includes("const MAIN_SLOT = 'main'"), true);
  assert.equal(client.includes('ctx.slots.inject(SIDEBAR_SLOT)'), true);
  assert.equal(client.includes('ctx.slots.inject(MAIN_SLOT)'), true);
  assert.equal(client.includes('key: PANEL_ID'), true);
});

test('usa la misma ruta de API que el servidor', () => {
  assert.equal(API_PREFIX, '/lifecode-rag/api');
  assert.equal(client.includes("const API = '/lifecode-rag/api'"), true);
});

test('no inventa una aprobación desde la interfaz', () => {
  assert.equal(/approval|allowed-once/.test(client), false);
  assert.equal(client.includes("action: 'forget'"), true);
  assert.equal(client.includes('confirm: true'), true, 'las acciones destructivas son explícitas');
});

test('un perfil Project-Aware se explica como panel no disponible, no como 404 crudo', () => {
  // Regresión de la primera versión estable: en un perfil PA la API v0.2 no se monta, y el panel
  // mostraba «Not Found (http-404)» como si el producto estuviera roto.
  assert.equal(client.includes('response.status === 404'), true);
  assert.equal(client.includes("unavailable.code = 'panel-unavailable'"), true);
  assert.equal(/Project-Aware: el panel del almacén v0\.2 no está disponible/.test(client), true);
  assert.equal(client.includes('`http-${response.status}`'), true, 'los demás errores conservan su código');
});
